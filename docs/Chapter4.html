# the code for the model is quite long and to do this exercise, we need to play around with values and re-run all of the code. => perfect opportunity for a function. 
cher <- read.csv("https://raw.githubusercontent.com/rmcelreath/rethinking/refs/heads/master/data/cherry_blossoms.csv", sep = ";")


do_splines <- function(n_knots = 15, 
                       width_prior = 10){
    
  # following the code from Solomon Kurz: https://bookdown.org/content/4857/geocentric-models.html#splines.
  
  d2 <-
    cher |> 
    drop_na(doy) # drop incomplete lines
  
  num_knots <- n_knots # this needs to be played around with
  knot_list <- quantile(d2$year, probs = seq(from = 0, to = 1, length.out = num_knots)) # placing the knots at data-informed locations
  
  B <- splines::bs(d2$year,
          knots = knot_list[-c(1, num_knots)], 
          degree = 3, 
          intercept = TRUE) # make the knots
  
  
  d3 <-
    d2 |> 
    mutate(B = B) # add the knots to the data-frame
  
  # some wrangling
  
  n_cols <- n_knots +2
  b <-
    B  |>  
    data.frame() |>  
    set_names(str_c(0, 1:9), 10:n_cols) |>   
    bind_cols(select(d2, year))  |>  
    pivot_longer(-year,
                 names_to = "bias_function",
                 values_to = "bias")
  
  
  
  # fit the model
  local_width_prior <- as.character(width_prior)
  b4.8 <- 
    brm(data = d3,
        family = gaussian,
        doy ~ 1 + B,
        prior = c(prior(normal(100, 10), class = Intercept),
                  prior_string(paste0("normal(0, ",local_width_prior,")"), class = "b"), # these are the widths of the prior, that should be played with for the exercise, therefore, we define this prior as strings
                  prior(exponential(1), class = sigma)),
        iter = 2000, warmup = 1000, chains = 4, cores = 4,
        seed = 4)
  
  
  # visualize the spline
  
  f <- fitted(b4.8)  |>  
    data.frame()  |>  
    bind_cols(d2)  |>  
    
    ggplot(aes(x = year, y = doy, ymin = Q2.5, ymax = Q97.5)) + 
    geom_vline(xintercept = knot_list, color = "white", alpha = 1/2) +
    geom_hline(yintercept = fixef(b4.8)[1, 1], color = "white", linetype = 2) +
    geom_point(color = "#ffb7c5", alpha = 1/2) +
    geom_ribbon(fill = "white", alpha = 2/3) +
    labs(x = "year",
         y = "day in year") +
    theme_bw() +
    theme(panel.background = element_rect(fill = "#4f455c"),
          panel.grid = element_blank())
  
  return(f)
}

4H3.

Suppose a collegue of yours, who works on allometry, glances at the practice problems just above. Your colleague exclaims, “That’s silly. Everyone knows that it’s only the logarithm of body weight that scales with height!” Let’s take your colleague’s advice and see what happens.

a.1)

Model the relationship between height (cm) and the natural logarithm of weight (log-kg). Use the entire Howell1 data frame, all 544 rows, adults and non-adults. Can you interpret the resulting estimates?

=> The text only talks about using the natural log, no mentioning of centralization. First let’s illutrate the data and the log-function. The log-scale tells us about “orders of magnitude” (p.98 in rethinking). That is, an increase in the log-scale of weight of one equals to a doubling of the weight.

Now fit the model. We adapt the prior to make it give plausable values on the log(weight) scale. The additive part of the formula is:

mu = a + b * log(weight)

the priors for a and b must therefore be consistent with the log-transformation of the weight to generate average expected heights in cm. a indicates the hight of a person that has a log(weight) = 0, which corresponds to a weight of 1: log(1) = : 0.

For the intercept a, we use a uniform(20, 100) prior, which corresponds to a height of for a person with a weight of 1 kg (a person who’s log(weight) = 0).

For the slope b, we follow along with the normal(0,10).

h3_fit <- brm(height ~ 1 + log_weight,

do_splines <- function(n_knots = 15, 
                       width_prior = 10){
    
  # following the code from Solomon Kurz: https://bookdown.org/content/4857/geocentric-models.html#splines.
  
  d2 <-
    cher |> 
    drop_na(doy) # drop incomplete lines
  
  num_knots <- n_knots # this needs to be played around with
  knot_list <- quantile(d2$year, probs = seq(from = 0, to = 1, length.out = num_knots)) # placing the knots at data-informed locations
  
  B <- splines::bs(d2$year,
          knots = knot_list[-c(1, num_knots)], 
          degree = 3, 
          intercept = TRUE) # make the knots
  
  
  d3 <-
    d2 |> 
    mutate(B = B) # add the knots to the data-frame
  
  # some wrangling
  
  n_cols <- n_knots +2
  b <-
    B  |>  
    data.frame() |>  
    set_names(str_c(0, 1:9), 10:n_cols) |>   
    bind_cols(select(d2, year))  |>  
    pivot_longer(-year,
                 names_to = "bias_function",
                 values_to = "bias")
  
  
  
  # fit the model
  local_width_prior <- as.character(width_prior)
  b4.8 <- 
    brm(data = d3,
        family = gaussian,
        doy ~ 1 + B,
        prior = c(prior(normal(100, 10), class = Intercept),
                  prior_string(paste0("normal(0, ",local_width_prior,")"), class = "b"), # these are the widths of the prior, that should be played with for the exercise, therefore, we define this prior as strings
                  prior(exponential(1), class = sigma)),
        iter = 2000, warmup = 1000, chains = 4, cores = 4,
        seed = 4)

  f <- fitted(b4.8)  |>  
    data.frame()  |>  
    bind_cols(d2)  |>  
    
    ggplot(aes(x = year, y = doy, ymin = Q2.5, ymax = Q97.5)) +

Now fit the model. We adapt the prior to make it give plausable values on the log(weight) scale. The additive part of the formula is: 

mu = a + b * log(weight)

the priors for a and b must therefore be consistent with the log-transformation of the weight to generate average expected heights in cm. `a` indicates the hight of a person that has a log(weight) = 0, which corresponds to a weight of 1: log(1) = : `r log(1)`. 

For the intercept `a`, we use a uniform(20, 100) prior, which corresponds to a height of for a person  with a weight of 1 kg (a person who's `log(weight) = 0`). 

For the slope `b`, we follow along with the normal(0,10).
